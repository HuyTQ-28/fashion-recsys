# Create volume
modal volume create my-volume

# Put file or folder to modal
modal volume put my-volume ./images /images

# Check files in a folder
modal volume ls my-volume /images

# Get file from modal
modal volume get my-volume /images/010/0108775044.jpg ./images/010/0108775044.jpg

# Remove volume
modal volume rm my-volume
